 669     fun play(movie: MovieEntity, startPositionMs: Long = 0) {
 670         viewModelScope.launch {
 671             val pid = currentProfileId()
 672             if (pid != null && !tv.own.owntv.core.content.AdultCategoryClassifier.allows(pid, movie.categoryId, profileDao, categoryDao)) return@launch
 673             // External player (global toggle): hand the stream URL to an external app and skip the
 674             // in-app engine entirely. History is still recorded (recently-watched); resume position
 675             // and the playing-movie HUD/progress tick are intentionally not — the external app owns
 676             // playback and OwnTV can't observe it.
 677             // A protected item cannot go to an external player: no standard intent extra
 678             // carries a licence URL, so the other app would open it and fail on the first segment.
 679             // Play it here instead, where the licence request can actually be made.
 680             if (settings.externalPlayerMovies.first() && movie.drmConfig == null) {
 681                 Log.d(TAG, "play movieId=${movie.id} -> external player")
 682                 val url = resolvedUrlOrNull(movie) ?: return@launch
 683                 val source = sourceDao.getById(movie.sourceId)
 684                 externalPlayerLauncher.launch(
 685                     url = url,
 686                     title = movie.name,
 687                     userAgent = source?.userAgent,
 688                     httpHeaders = SourceOverrides.headersWithReferer(movie.httpHeaders, source),
 689                     startPositionMs = startPositionMs, // the screen's Resume / Start over answer
 690                 )
 691                 if (pid != null) {
 692                     runCatching {
 693                         historyDao.record(WatchHistoryEntity(profileId = pid, mediaType = MediaType.MOVIE, itemId = movie.id))
 694                     }.onFailure { t -> Log.w(TAG, "external play history record failed movieId=${movie.id} profile=$pid", t) }
 695                 }
 696                 return@launch
 697             }
 698             val source = sourceDao.getById(movie.sourceId)
 699             val sourceUa = source?.userAgent
 700             // Stalker: mint the playable URL right before playback (create_link, type=vod). The stored
 701             // streamUrl stays the cmd — it's the item's identity (engine pins, downloads, history).
 702             val playUrl = if (streamUrlResolver.needsResolve(source)) {
 703                 try {
 704                     streamUrlResolver.resolve(source!!, movie.streamUrl, vod = true)
 705                 } catch (e: Exception) {
 706                     Log.w(TAG, "stalker resolve failed movieId=${movie.id}", e)
 707                     return@launch
 708                 }
 709             } else {
 710                 movie.streamUrl
 711             }
 712             Log.d(TAG, "play movieId=${movie.id} profile=$pid startPositionMs=$startPositionMs")
 713             val pinKey = tv.own.owntv.core.player.enginePinKey(movie.sourceId, "MOVIE", movie.remoteId)
 714             player.play(
 715                 playUrl,
 716                 title = movie.name,
 717                 year = movie.year?.toString(),
 718                 isLive = false,
 719                 startPositionMs = startPositionMs,
 720                 userAgent = sourceUa,
 721                 httpHeaders = SourceOverrides.headersWithReferer(movie.httpHeaders, source),
 722                 vodEngineOverride = SourceOverrides.vodEngineOf(source),
 723                 drmConfig = movie.drmConfig,
 724                 manifestType = movie.manifestType,
 725                 // Engine pins key on this, not on playUrl (a Stalker playUrl is minted per play).
 726                 contentKey = pinKey,
 727                 // A Stalker create_link URL dies before a long film ends; give the player a way to
 728                 // mint a fresh one instead of retrying the expired link. Null for M3U/Xtream, which also
 729                 // clears any provider the previous item left on the player.
 730                 reconnectProvider = if (streamUrlResolver.needsResolve(source)) {
 731                     tv.own.owntv.core.stalker.ReconnectUrlProvider {
 732                         runCatching { streamUrlResolver.resolve(source!!, movie.streamUrl, vod = true) }
 733                             .onFailure { Log.w(TAG, "stalker VOD reconnect resolve failed movieId=${movie.id}", it) }
 734                             .getOrNull()
 735                     }
 736                 } else null,
 737             )
 738             _playingMovie.value = movie
 739             playingRef = pid?.let { PlayingRef(movie, it, pinKey) }
 740             // Enable the player's OpenSubtitles search for this movie. tmdbId is
 741             // resolved from the metadata cache when available for a stronger match.
 742             if (pid != null) {
 743                 val tmdbId = runCatching { metadata.resolveMovie(movie)?.tmdbId?.toLong() }.getOrNull()
 744                 subtitleController.setMovie(pid, movie, tmdbId)
 745             }
 746             if (pid != null) {
 747                 runCatching {
 748                     historyDao.record(WatchHistoryEntity(profileId = pid, mediaType = MediaType.MOVIE, itemId = movie.id))
 749                 }.onFailure { t ->
 750                     Log.w(TAG, "play history record failed movieId=${movie.id} profile=$pid", t)
 751                 }
 752             }
 753         }
 754     }

 284     private fun interpreter(): String? = listOf("python", "python3").firstOrNull { candidate ->
 285         runCatching {
 286             execOps.exec {
 287                 commandLine(candidate, "--version")
 288                 isIgnoreExitValue = true
 289                 standardOutput = ByteArrayOutputStream()
 290                 errorOutput = ByteArrayOutputStream()
 291             }.exitValue == 0
 292         }.getOrDefault(false)
 293     }

 295     @TaskAction
 296     fun verify() {
 297         val python = interpreter()
 298         if (python == null) {
 299             // Failing here would block anyone without Python from building at all. Warn loudly
 300             // instead — CI still enforces it, so the worst case is a late failure, not a missed one.
 301             logger.warn(
 302                 "\n  WARNING: Python was not found, so the hardcoded-text check did not run." +
 303                     "\n  Install Python 3 to catch untranslatable text before pushing; CI will still catch it.\n",
 304             )
 305             stamp.get().asFile.writeText("skipped: no python interpreter\n")
 306             return
 307         }
 308         val output = ByteArrayOutputStream()
 309         val result = execOps.exec {
 310             workingDir = repoRoot.get().asFile
 311             commandLine(python, "tools/i18n/check_hardcoded_strings.py", "verify", "--bootstrap")
 312             environment("PYTHONIOENCODING", "utf-8")
 313             isIgnoreExitValue = true
 314             standardOutput = output
 315             errorOutput = output
 316         }
 317         if (result.exitValue != 0) {
 318             logger.error(output.toString(Charsets.UTF_8))
 319             throw GradleException("Hardcoded text check failed — see the report above.")
 320         }
 321         stamp.get().asFile.writeText("ok\n")
 322     }
