 386: /** Scroll only as far as the focused item, plus one row of room, needs (the TV default pins it a third down). */

 388: internal val edgeScrollSpec = object : androidx.compose.foundation.gestures.BringIntoViewSpec {
 389:     override fun calculateScrollDistance(offset: Float, size: Float, containerSize: Float): Float {
 390:         // One row of room kept ahead of the highlight, so the next row is already laid out: a held
 391:         // ▲/▼ otherwise outran the list at the edge and stopped.
 392:         val room = size.coerceAtMost((containerSize - size).coerceAtLeast(0f) / 2f)
 393:         val lead = offset - room
 394:         val trailing = offset + size + room - containerSize
 395:         return when {
 396:             lead >= 0f && trailing <= 0f -> 0f
 397:             lead < 0f || size > containerSize -> lead
 398:             else -> trailing
 399:         }
 400:     }
