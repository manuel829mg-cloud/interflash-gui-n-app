1537     /** Go full-screen on [channel]. Cancels any pending direct-tune zap rebuild first, so normal
1538      *  navigation always wins over an in-flight rebuild. ExoPlayer is the **primary** live engine
1539      *  (instant for HLS, and it plays the channels mpv struggles to open): promote the running
1540      *  preview if it's already this channel, else (re)start ExoPlayer on it. We fall back to the
1541      *  full **mpv** player ONLY if ExoPlayer **errors** (a stream it can't open) — never just
1542      *  because it's still loading (clicking OK before the preview is ready used to drop to mpv and
1543      *  stick on a black screen for HLS). */
1544     fun ensurePlaying(channel: ChannelEntity) {
1545         // A deliberate pick, not a CH+/- step: an unclaimed zap press must not be timed against it.
1546         tv.own.owntv.player.ZapTiming.shared.clear()
1547         startTune(channel)
1548     }

1550     /** [ensurePlaying] without forgetting the zap press: the tune a CH+/- step opens, timed from that press. */
1551     private fun startTune(channel: ChannelEntity) {
1552         zapList.cancelPendingRebuild()
1553         // A deliberate pick supersedes a CH+/- step still waiting out its delay — otherwise the deferred
1554         // tune would land half a second later and drag the user off the channel they just chose.
1555         pendingZapTuneJob?.cancel()
1556         pendingZapTuneJob = null
1557         // Before the full-screen player is first drawn: a channel ExoPlayer is already previewing opens on
1558         // ExoPlayer's surface straight away, instead of on mpv's until the tune below gets there.
1559         live.expectPromotion(channel)
1560         // The controller's one tune job: a quicker second pick cancels the first wherever it has
1561         // got to, so two picks can no longer finish in the wrong order.
1562         live.launch { playChannel(channel) }
1563     }

1568     /** Internal playback: the canonical ExoPlayer / mpv / Stalker / history side-effects for a
1569      *  channel. Direct-tune's background rebuild path calls this without cancelling the rebuild
1570      *  so the in-flight rebuild it owns isn't killed by its own play. */
1571     private suspend fun playChannel(channel: ChannelEntity) {
1572         val pid = currentProfileId() ?: return
1573         if (!tv.own.owntv.core.content.AdultCategoryClassifier.allows(pid, channel.categoryId, profileDao, categoryDao)) return
1574         // Live TV set to play externally: hand the channel over instead of tuning an in-app engine.
1575         // History is still recorded, so the channel shows up in History/Recently watched either way.
1576         // A protected channel stays in-app whatever this setting says: no standard intent extra
1577         // carries a licence URL, so the external player would open it and fail immediately.
1578         if (externalPlayerOn.value && channel.drmConfig == null) { playExternal(channel); return }
1579         _previewChannel.value = channel
1580         clearTimeshift() // normal live = not timeshifted
1581         _catchupActive.value = false // tuning live ends any archive playback the HUD was showing
1582         // Core routes it (DRM, the pin, a learned panel refusal, the playlist, the setting), arms the
1583         // ladder and starts the engine — the same code the phone runs.
1584         live.start(channel, getSource(channel.sourceId))
1585         recordLiveHistory(channel)
1586     }
